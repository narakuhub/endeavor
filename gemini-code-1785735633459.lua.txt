			table.remove(reg, i)
		end
	end

			if debug.info(to_hook, "s") == "[C]" then

local function ExecuteNarakuBypass()

	local adonisFound = false

	-- 1. Scan Threads
	if getallthreads then
		for _, v in next, getallthreads() do
			local source = debug.info(v, 1, "s")
			if source and (source:find(".Core.Anti", 1, true) or source:find(".Plugins.Anti_Cheat", 1, true)) then
				table.insert(threads, v)
				adonisFound = true
			end
		end
	end

	-- 2. Kill Anti-Cheat Threads
	for _, v in next, threads do
		pcall(task.cancel, v)
	end

	-- 3. Hook Detection Functions (GC Hooking)
	if hookfunction then
		if filtergc then
			local success, tables = pcall(function()
				return filtergc("table", {Keys = {"RLocked", "Detected"}}, true)
			end)
			if success and tables then
				for _, v in next, tables do
					if type(v) == "function" and not isfunctionhooked(v) then
						hookfunction(v, coroutine.yield)
					end
				end
			end
		elseif getgc then
			local success, gc = pcall(function() return getgc(true) end)
			if success and gc then
				for _, v in next, gc do
					if type(v) == "table" and rawget(v, "Detected") and rawget(v, "RLocked") then
						for _, fn in next, v do
							if type(fn) == "function" and not isfunctionhooked(fn) then
								hookfunction(fn, coroutine.yield)
							end
						end
						break
					end
				end
			end
		end
	end

	if adonisFound then
		print("[NARAKU BYPASS]: Adonis Anti-Cheat Bypassed Successfully!")
	else
		print("[NARAKU BYPASS]: Anti-Cheat bypassed / No active threads detected.")
	end

	return true
end
